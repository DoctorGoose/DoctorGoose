import matplotlib.pyplot as plt import numpy as np # Example: a single-time plot of a 440 Hz sine wave # For a "live" approach, you'd dynamically re-run this code # with updated frequency/waveform from JavaScript. sr = 44100 duration = 0.01 # 10 ms for a short snippet t = np.linspace(0, duration, int(sr*duration), endpoint=False) freq = 440 waveform = np.sin(2 * np.pi * freq * t) plt.figure(figsize=(8, 3)) plt.plot(t[:1000], waveform[:1000]) plt.xlabel("Time (s)") plt.ylabel("Amplitude") plt.title("Example Waveform (440 Hz Sine)") plt.tight_layout() plt.show()
